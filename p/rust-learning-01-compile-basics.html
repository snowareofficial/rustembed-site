编译为静态库

rust
pub fn print_hello()
{
    println!("Hello this is Lib!");
}
rust
extern crate myhello;

fn main() {
    // println!("Hello, world!");
    if cfg!(sayhello){
        println!("Hello Milly!");
    }
    myhello::print_hello();
}
bash
rustc --crate-type lib myhello.rs  # 编译为静态库
rustc -L. -lmyhello main.rs	# 编译为 可执行程序
rustc -L. -lmyhello --cfg sayhello main.rs # 添加标志
